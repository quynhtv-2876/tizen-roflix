## Objective

Polish the movie details view in `roflix/index.html`:
- Server tabs navigable by LEFT/RIGHT D-pad
- Episode list navigable by UP/DOWN
- Clear focus indicators on both tabs and episodes
- BACK button returns to grid correctly
- Description truncation (300 chars + "...")

---

## Current View Structure

The `details-view` should contain:
1. Left panel: Movie poster + title + description
2. Right panel: Server tabs (top) + Episode list (bottom)

The D-pad focus flow should be:
- Default: focus on episode list
- LEFT/RIGHT on episode list area: switches server tab
- UP/DOWN: moves between episodes
- ENTER on episode: plays video
- BACK: returns to grid

### Step 2: Ensure renderDetails() populates tabs

```javascript
function renderDetails(movieData) {
  // Poster & text
  document.getElementById('details-poster').src = movieData.imageUrl;
  document.getElementById('details-title').textContent = movieData.title;
  document.getElementById('details-original-title').textContent =
    movieData.origin_title ? `(${movieData.origin_title})` : '';

  const desc = movieData.description || '';
  document.getElementById('details-description').textContent =
    desc.length > 300 ? desc.slice(0, 300) + '…' : desc;

  // Server tabs
  const tabsEl = document.getElementById('server-tabs');
  tabsEl.innerHTML = '';
  (movieData.episodes || []).forEach((server, idx) => {
    const tab = document.createElement('button');
    tab.className = 'server-tab' + (idx === 0 ? ' active' : '');
    tab.textContent = server.server_name || `Server ${idx + 1}`;
    tab.dataset.serverIndex = idx;
    tabsEl.appendChild(tab);
  });

  // Episodes for first server
  episodeServerIndex = 0;
  episodeIndex = 0;
  renderEpisodes(movieData.episodes[0]?.episodes || []);
}

function renderEpisodes(episodes) {
  const listEl = document.getElementById('episode-list');
  listEl.innerHTML = '';
  (episodes || []).forEach((ep, idx) => {
    const item = document.createElement('div');
    item.className = 'episode-item' + (idx === 0 ? ' focused' : '');
    item.textContent = ep.name;
    item.dataset.episodeIndex = idx;
    listEl.appendChild(item);
  });
}
```

### Step 3: Update server tab switching (LEFT/RIGHT in details view)

```javascript
// In keydown handler, when currentView === 'details':
case KEY_CODES.LEFT:
  if (episodeServerIndex > 0) {
    episodeServerIndex--;
    episodeIndex = 0;
    updateServerTabFocus();
    renderEpisodes(episodesData[episodeServerIndex].episodes);
  }
  break;

case KEY_CODES.RIGHT:
  if (episodeServerIndex < episodesData.length - 1) {
    episodeServerIndex++;
    episodeIndex = 0;
    updateServerTabFocus();
    renderEpisodes(episodesData[episodeServerIndex].episodes);
  }
  break;

case KEY_CODES.UP:
  if (episodeIndex > 0) {
    episodeIndex--;
    updateEpisodeFocus();
  }
  break;

case KEY_CODES.DOWN:
  if (episodeIndex < episodesData[episodeServerIndex].episodes.length - 1) {
    episodeIndex++;
    updateEpisodeFocus();
  }
  break;

case KEY_CODES.ENTER:
  playSelectedEpisode();
  break;

case KEY_CODES.BACK:
  switchView('grid');
  break;
```

### Step 4: Add updateServerTabFocus() and updateEpisodeFocus() helpers

```javascript
function updateServerTabFocus() {
  document.querySelectorAll('.server-tab').forEach((tab, idx) => {
    tab.classList.toggle('active', idx === episodeServerIndex);
    tab.classList.toggle('focused', idx === episodeServerIndex);
  });
}

function updateEpisodeFocus() {
  const items = document.querySelectorAll('.episode-item');
  items.forEach((item, idx) => {
    item.classList.toggle('focused', idx === episodeIndex);
  });
  // Auto-scroll into view
  items[episodeIndex]?.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
}
```
