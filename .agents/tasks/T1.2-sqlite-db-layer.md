## Objective

Create `roflix-server/db.js` — a SQLite database layer using `better-sqlite3` that provides:
- Movie list cache (per batch)
- Episode metadata cache (per slug)
- Watch history (per slug + episode)

---

### Step 2: Create roflix-server/db.js

Create the file with this implementation:

```javascript
'use strict';

const Database = require('better-sqlite3');
const path = require('path');
const fs = require('fs');

// Ensure data directory exists
const DATA_DIR = path.join(__dirname, 'data');
if (!fs.existsSync(DATA_DIR)) {
  fs.mkdirSync(DATA_DIR, { recursive: true });
}

const DB_PATH = path.join(DATA_DIR, 'roflix.db');
const db = new Database(DB_PATH);

// Enable WAL mode for better performance
db.pragma('journal_mode = WAL');

// ─── Schema ──────────────────────────────────────────────────────────────────

db.exec(`
  CREATE TABLE IF NOT EXISTS movie_cache (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    batch      INTEGER NOT NULL,
    slug       TEXT    NOT NULL,
    title      TEXT,
    poster_url TEXT,
    cached_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(batch, slug)
  );

  CREATE TABLE IF NOT EXISTS episode_cache (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    movie_slug   TEXT    NOT NULL UNIQUE,
    title        TEXT,
    origin_title TEXT,
    description  TEXT,
    image_url    TEXT,
    episodes_json TEXT,
    cached_at    DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS watch_history (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    movie_slug   TEXT NOT NULL,
    episode_name TEXT NOT NULL,
    position     INTEGER NOT NULL DEFAULT 0,
    updated_at   DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(movie_slug, episode_name)
  );
`);

// ─── Constants ───────────────────────────────────────────────────────────────

const CACHE_TTL_MS = 60 * 60 * 1000; // 1 hour

// ─── Movie Cache ─────────────────────────────────────────────────────────────

/**
 * Get cached movies for a batch.
 * Returns [] if cache is missing or expired (> 1 hour old).
 * @param {number} batch
 * @returns {{ slug: string, title: string, imageUrl: string, link: string }[]}
 */
function getMoviesCache(batch) {
  const rows = db.prepare(`
    SELECT slug, title, poster_url, cached_at
    FROM movie_cache
    WHERE batch = ?
    ORDER BY id ASC
  `).all(batch);

  if (!rows.length) return [];

  // Check TTL using first row's cached_at
  const cachedAt = new Date(rows[0].cached_at).getTime();
  if (Date.now() - cachedAt > CACHE_TTL_MS) {
    console.log(`[DB] Cache expired for batch ${batch} — clearing`);
    db.prepare('DELETE FROM movie_cache WHERE batch = ?').run(batch);
    return [];
  }

  return rows.map(row => ({
    slug: row.slug,
    title: row.title,
    imageUrl: row.poster_url,
    link: `/phim/${row.slug}`,
  }));
}

/**
 * Save movies to cache for a batch.
 * Replaces any existing cache for this batch.
 * @param {{ slug: string, title: string, imageUrl: string }[]} movies
 * @param {number} batch
 */
function cacheMovies(movies, batch) {
  // Clear existing batch cache first
  db.prepare('DELETE FROM movie_cache WHERE batch = ?').run(batch);

  const insert = db.prepare(`
    INSERT OR REPLACE INTO movie_cache (batch, slug, title, poster_url)
    VALUES (?, ?, ?, ?)
  `);

  const insertMany = db.transaction((movies) => {
    for (const movie of movies) {
      insert.run(batch, movie.slug, movie.title, movie.imageUrl);
    }
  });

  insertMany(movies);
  console.log(`[DB] Cached ${movies.length} movies for batch ${batch}`);
}

// ─── Episode Cache ────────────────────────────────────────────────────────────

/**
 * Get cached episode data for a movie slug.
 * Returns null if cache is missing or expired.
 * @param {string} slug
 * @returns {object|null}
 */
function getEpisodeCache(slug) {
  const row = db.prepare(`
    SELECT title, origin_title, description, image_url, episodes_json, cached_at
    FROM episode_cache
    WHERE movie_slug = ?
  `).get(slug);

  if (!row) return null;

  const cachedAt = new Date(row.cached_at).getTime();
  if (Date.now() - cachedAt > CACHE_TTL_MS) {
    console.log(`[DB] Episode cache expired for slug: ${slug}`);
    db.prepare('DELETE FROM episode_cache WHERE movie_slug = ?').run(slug);
    return null;
  }

  return {
    title: row.title,
    origin_title: row.origin_title,
    description: row.description,
    imageUrl: row.image_url,
    episodes: JSON.parse(row.episodes_json),
  };
}

/**
 * Save episode data for a movie slug.
 * @param {string} slug
 * @param {object} movieData - { title, origin_title, description, imageUrl, episodes }
 */
function cacheEpisodes(slug, movieData) {
  db.prepare(`
    INSERT OR REPLACE INTO episode_cache
      (movie_slug, title, origin_title, description, image_url, episodes_json, cached_at)
    VALUES (?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP)
  `).run(
    slug,
    movieData.title,
    movieData.origin_title || null,
    movieData.description || null,
    movieData.imageUrl || null,
    JSON.stringify(movieData.episodes),
  );
  console.log(`[DB] Cached episodes for slug: ${slug}`);
}

// ─── Watch History ────────────────────────────────────────────────────────────

/**
 * Save (upsert) playback position for a movie episode.
 * @param {string} movieSlug
 * @param {string} episodeName
 * @param {number} position - seconds elapsed
 */
function saveWatchPosition(movieSlug, episodeName, position) {
  db.prepare(`
    INSERT INTO watch_history (movie_slug, episode_name, position, updated_at)
    VALUES (?, ?, ?, CURRENT_TIMESTAMP)
    ON CONFLICT(movie_slug, episode_name)
    DO UPDATE SET position = excluded.position, updated_at = CURRENT_TIMESTAMP
  `).run(movieSlug, episodeName, position);
}

/**
 * Get saved playback position for a movie episode.
 * @param {string} movieSlug
 * @param {string} episodeName
 * @returns {number} seconds elapsed, or 0 if not found
 */
function getWatchPosition(movieSlug, episodeName) {
  const row = db.prepare(`
    SELECT position FROM watch_history
    WHERE movie_slug = ? AND episode_name = ?
  `).get(movieSlug, episodeName);

  return row ? row.position : 0;
}

// ─── Exports ─────────────────────────────────────────────────────────────────

module.exports = {
  getMoviesCache,
  cacheMovies,
  getEpisodeCache,
  cacheEpisodes,
  saveWatchPosition,
  getWatchPosition,
};
```

## Notes

- `better-sqlite3` is **synchronous** (intentionally) — Express handles concurrency at the OS level. No need for async/await in db.js.
- WAL mode (`journal_mode = WAL`) enables concurrent reads while writing.
- `UNIQUE(batch, slug)` prevents duplicate entries when cache is refreshed.
